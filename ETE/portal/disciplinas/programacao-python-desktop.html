🔀 Estruturas de Controle

Condicional (if/elif/else)

Python # Sistema de notas nota = float(input("Nota: ")) if nota >= 7.0: situacao = "Aprovado ✅" elif nota >= 5.0: situacao = "Recuperação ⚠️" else: situacao = "Reprovado ❌" print(f"Situação: {situacao}")

Repetição (for / while)

Python # FOR — quando sabe quantas vezes alunos = ["Ana", "João", "Maria"] for aluno in alunos: print(f"Chamada: {aluno}") # WHILE — quando depende de condição tentativas = 0 while tentativas < 3: senha = input("Senha: ") if senha == "1234": print("Acesso liberado!") break tentativas += 1
⚠️ Indentação é obrigatória em Python! Use sempre 4 espaços. Sem { } — a indentação define os blocos.

🔧 Funções e Módulos

Funções organizam o código em blocos reutilizáveis. Módulos agrupam funções relacionadas.

Python # validacoes.py — Módulo de validação def validar_cpf(cpf: str) -> bool: """Valida CPF com 11 dígitos.""" cpf = cpf.replace(".", "").replace("-", "") if len(cpf) != 11: return False if cpf == cpf[0] * 11: return False return True def validar_email(email: str) -> bool: """Verifica se email contém @ e .""" return "@" in email and "." in email def validar_idade(idade: int) -> bool: """Idade entre 0 e 130.""" return 0 <= idade <= 130
Python # main.py — Importando o módulo from validacoes import ( validar_cpf, validar_email, validar_idade ) cpf = input("CPF: ") if validar_cpf(cpf): print("✅ CPF válido") else: print("❌ CPF inválido") email = input("Email: ") if validar_email(email): print("✅ Email válido") else: print("❌ Email inválido")
💡 Cada arquivo .py é um módulo. import permite reutilizar em qualquer projeto!

Python import tkinter as tk from tkinter import ttk, messagebox class CadastroApp: def __init__(self, root): root.title("Cadastro de Aluno") root.geometry("400x300") ttk.Label(root, text="Nome:").pack(pady=5) self.nome = ttk.Entry(root, width=40) self.nome.pack() ttk.Label(root, text="Email:").pack(pady=5) self.email = ttk.Entry(root, width=40) self.email.pack() ttk.Button(root, text="Salvar", command=self.salvar).pack(pady=20) def salvar(self): messagebox.showinfo("Sucesso", f"Aluno {self.nome.get()} cadastrado!") root = tk.Tk() app = CadastroApp(root) root.mainloop()

Mesma operação de banco em 3 linguagens

Python import psycopg2 # Conexão com PostgreSQL conn = psycopg2.connect( host="localhost", database="biblioteca", user="postgres", password="postgres" ) cursor = conn.cursor() # INSERT cursor.execute(""" INSERT INTO livros (titulo, autor, ano) VALUES (%s, %s, %s) """, ("Dom Casmurro", "Machado de Assis", 1899)) conn.commit() # SELECT cursor.execute("SELECT * FROM livros") livros = cursor.fetchall() for livro in livros: print(f"{livro[1]} — {livro[2]}") conn.close()
TypeScript // database.ts (Node.js + pg) import { Pool } from 'pg'; const pool = new Pool({ host: 'localhost', database: 'biblioteca', user: 'postgres', password: 'postgres' }); // INSERT await pool.query( `INSERT INTO livros (titulo, autor, ano) VALUES ($1, $2, $3)`, ['Dom Casmurro', 'Machado de Assis', 1899] ); // SELECT const { rows } = await pool.query( 'SELECT * FROM livros' ); rows.forEach(livro => console.log(`${livro.titulo} — ${livro.autor}`) );
